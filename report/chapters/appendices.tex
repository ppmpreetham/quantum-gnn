\chapter{Pseudo-Code of the Pipeline, the Solver and the Motion Engine}
\label{app:pseudo}

This appendix gives, in algorithmic form, the per-cycle pipeline of
Chapter~\ref{ch:arch} and the projected edge annealing of
Chapter~\ref{ch:solvers}. The relaxation ladder and the motion engine's step
are described in Sections~\ref{sec:sol:ladder} and~\ref{ch:engine}.

\begin{algorithm}[H]
\caption{One pipeline cycle}
\label{alg:app:cycle}
\begin{algorithmic}[1]
\Require telemetry for cycle $t$, trust $T^{t-1}$, previous kept edges
\State build $G_t$ and the edge and rover features, using $T^{t-1}$
\State prune with projected annealing and the relaxation ladder (Section~\ref{sec:sol:ladder})
\State run two layers of trust-weighted message passing on the kept graph
\State score every rover for every task with gates, criticality and delivery factor
\State assign tasks greedily under per-rover capacity; explore with probability $\varepsilon$
\State observe execution and delivery outcomes
\State update trust by EWMA for assigned rovers and by aging for the others
\Ensure kept edges, assignment and $T^{t}$
\end{algorithmic}
\end{algorithm}

\begin{algorithm}[H]
\caption{Projected edge annealing}
\label{alg:app:pea}
\begin{algorithmic}[1]
\Require QUBO builder, warm edge set $W$, schedule $(\tau_0,\tau_f,\gamma)$, steps $k$, restarts $n_r$
\State best $\gets$ none
\For{restart $=1,\dots,n_r$}
  \State $S\gets W$ on the first restart, otherwise a degree-cap-aware random spanning tree
  \If{$S$ cannot be projected to feasibility} \State \textbf{continue} \EndIf
  \State $\tau\gets\tau_0$
  \While{$\tau>\tau_f$}
    \For{$k$ times}
      \State propose $S'$: toggle one optional edge, or swap one kept and one unkept edge
      \If{$S'$ projects to feasibility (closed-form slacks, spanning-tree flow)}
        \State accept with probability $\min\{1,\exp(-(H(S')-H(S))/\tau)\}$
      \EndIf
    \EndFor
    \State $\tau\gets\gamma\,\tau$
  \EndWhile
  \State polish: alternate single-toggle and pairwise-swap passes to a fixed point
  \State keep $S$ if it is the best so far
\EndFor
\If{best is none} \State \textbf{fail}: no feasible edge set \EndIf
\Ensure best edge set and its feasible bit vector
\end{algorithmic}
\end{algorithm}
